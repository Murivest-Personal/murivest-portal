   108→export async function investorList(
   109→  actx: AdminContext,
   110→  opts: {search?: string; tier?: string; verification?: string; classification?: string; dormant?: boolean; limit?: number} = {},
   111→): Promise<InvestorListRow[]> {
   112→  const scope = investorScopeWhere(actx)
   113→  const search = opts.search?.trim().toLowerCase() || null
   114→
   115→  let profileQuery = supabase.from('investor_profiles').select('*')
   116→  if (scope) profileQuery = profileQuery.eq('advisor_id', scope.advisorId)
   117→  if (opts.tier) profileQuery = profileQuery.eq('access_tier', opts.tier)
   118→  if (opts.verification) profileQuery = profileQuery.eq('verification_status', opts.verification)
   119→  if (opts.classification) profileQuery = profileQuery.eq('classification', opts.classification)
   120→  const profileRows = unwrap<InvestorProfileRow[]>(
   121→    await profileQuery.order('created_at', {ascending: false}).limit(opts.limit ?? 500),
   122→    'profiles.list',
   123→  )
   124→
   125→  // relation include → separate query + JS join
   126→  const profileUserIds = profileRows.map((p) => p.user_id)
   127→  const userRows = profileUserIds.length
   128→    ? unwrap<UserRow[]>(await supabase.from('users').select('*').in('id', profileUserIds), 'users.list')
   129→    : []
   130→  const userBy = new Map(userRows.map((u) => [u.id, u]))
   131→
   132→  let rows = profileRows.filter((p) => {
   133→    if (!search) return true
   134→    const u = userBy.get(p.user_id)
   135→    return (
   136→      (u?.email ?? '').toLowerCase().includes(search) ||
   137→      (u?.full_name ?? '').toLowerCase().includes(search) ||
   138→      (p.country_residence ?? '').toLowerCase().includes(search)
   139→    )
   140→  })
   141→
   142→  if (opts.dormant) {
   143→    const cutoff = Date.now() - 60 * 86400000
   144→    rows = rows.filter((p) => {
   145→      const lastLogin = userBy.get(p.user_id)?.last_login_at
   146→      return !lastLogin || new Date(lastLogin).getTime() < cutoff
   147→    })
   148→  }
   149→
   150→  const userIds = rows.map((r) => r.user_id)
   151→  const [orderRows, revenueEventRows, mandateRows, orgRows] = await Promise.all([
   152→    userIds.length ? unwrap<OrderRow[]>(await supabase.from('orders').select('*').in('user_id', userIds), 'orders.list') : [],
   153→    userIds.length ? unwrap<RevenueEventRow[]>(await supabase.from('revenue_events').select('*'), 'revenue.list') : [],
   154→    userIds.length ? unwrap<MandateRow[]>(await supabase.from('mandates').select('*').in('user_id', userIds).eq('status', 'active'), 'mandates.list') : [],
   155→    unwrap<OrganisationRow[]>(await supabase.from('organisations').select('*'), 'organisations.list'),
   156→  ])
   157→  // revenue-event → order join: separate query + JS map
   158→  const revenueOrderRows = revenueEventRows.length
   159→    ? unwrap<OrderRow[]>(
   160→        await supabase.from('orders').select('*').in('id', [...new Set(revenueEventRows.map((e) => e.order_id))]),
   161→        'orders.list',
   162→      )
   163→    : []
   164→  const revenueOrderUser = new Map(revenueOrderRows.map((o) => [o.id, o.user_id]))
   165→
   166→  const orderCount = new Map<string, number>()
   167→  for (const o of orderRows) orderCount.set(o.user_id, (orderCount.get(o.user_id) ?? 0) + 1)
   168→  const revenueBy = new Map<string, number>()
   169→  for (const e of revenueEventRows) {
   170→    const uid = revenueOrderUser.get(e.order_id)
   171→    if (!uid) continue
   172→    revenueBy.set(uid, (revenueBy.get(uid) ?? 0) + e.gross_usd)
   173→  }
   174→  const mandateCount = new Map<string, number>()
   175→  for (const m of mandateRows) mandateCount.set(m.user_id, (mandateCount.get(m.user_id) ?? 0) + 1)
   176→  const orgNames = new Map(orgRows.map((o) => [o.id, o.name]))
   177→
   178→  return rows.map((p) => {
   179→    const u = userBy.get(p.user_id)
   180→    return {
   181→      userId: p.user_id,
   182→      email: u?.email ?? '',
   183→      fullName: u?.full_name ?? null,
   184→      status: u?.status ?? '',
   185→      classification: p.classification,
   186→      verificationStatus: p.verification_status,
   187→      kycStatus: p.kyc_status,
   188→      amlStatus: p.aml_status,
   189→      accessTier: p.access_tier,
   190→      ndaAccepted: p.nda_accepted,
   191→      advisorId: p.advisor_id,
   192→      source: p.source,
   193→      countryResidence: p.country_residence,
   194→      organisationId: p.organisation_id,
   195→      organisationName: p.organisation_id ? orgNames.get(p.organisation_id) ?? null : null,
   196→      equityAvailable: p.equity_available,
   197→      totalCapacity: p.total_capacity,
   198→      capitalFloor: p.capital_floor,
   199→      capitalCeiling: p.capital_ceiling,
   200→      capitalCurrency: p.capital_currency,
   201→      createdAt: new Date(u?.created_at ?? Date.now()),
   202→      lastLoginAt: u?.last_login_at ? new Date(u.last_login_at) : null,
   203→      orderCount: orderCount.get(p.user_id) ?? 0,
   204→      revenueUsd: Math.round((revenueBy.get(p.user_id) ?? 0) * 100) / 100,
   205→      activeMandates: mandateCount.get(p.user_id) ?? 0,
   206→    }
   207→  })
   208→}

   210→/** Organisation + members + member users (separate queries + JS join). */
   211→async function organisationWithMembers(organisationId: string) {
   212→  const orgRow = unwrapNullable<OrganisationRow>(
   213→    await supabase.from('organisations').select('*').eq('id', organisationId).maybeSingle(),
   214→    'organisations.get',
   215→  )
   216→  if (!orgRow) return null
   217→  const memberRows = unwrap<OrganisationMemberRow[]>(
   218→    await supabase.from('organisation_members').select('*').eq('organisation_id', organisationId),
   219→    'organisationMembers.list',
   220→  )
   221→  const memberUserIds = memberRows.map((m) => m.user_id)
   222→  const memberUserRows = memberUserIds.length
   223→    ? unwrap<UserRow[]>(await supabase.from('users').select('*').in('id', memberUserIds), 'users.list')
   224→    : []
   225→  const memberUserBy = new Map(memberUserRows.map((u) => [u.id, u]))
   226→  return {
   227→    ...toCamel(orgRow),
   228→    members: memberRows.map((m) => {
   229→      const u = memberUserBy.get(m.user_id)
   230→      return {
   231→        ...toCamel(m),
   232→        user: u ? {id: u.id, fullName: u.full_name, email: u.email} : {id: m.user_id, fullName: null, email: ''},
   233→      }
   234→    }),
   235→  }

   375→export async function organisationList() {
   376→  const orgRows = unwrap<OrganisationRow[]>(
   377→    await supabase.from('organisations').select('*').order('created_at', {ascending: false}),
   378→    'organisations.list',
   379→  )
   380→  // nested relation include (members + users) → separate queries + JS join
   381→  const orgIds = orgRows.map((o) => o.id)
   382→  const memberRows = orgIds.length
   383→    ? unwrap<OrganisationMemberRow[]>(
   384→        await supabase.from('organisation_members').select('*').in('organisation_id', orgIds),
   385→        'organisationMembers.list',
   386→      )
   387→    : []
   388→  const memberUserIds = [...new Set(memberRows.map((m) => m.user_id))]
   389→  const memberUserRows = memberUserIds.length
   390→    ? unwrap<UserRow[]>(await supabase.from('users').select('*').in('id', memberUserIds), 'users.list')
   391→    : []
   392→  const memberUserBy = new Map(memberUserRows.map((u) => [u.id, u]))
   393→  const membersByOrg = new Map<string, OrganisationMemberRow[]>()
   394→  for (const m of memberRows) {
   395→    const arr = membersByOrg.get(m.organisation_id) ?? []
   396→    arr.push(m)
   397→    membersByOrg.set(m.organisation_id, arr)
   398→  }
   399→  return orgRows.map((o) => ({
   400→    id: o.id,
   401→    name: o.name,
   402→    legalName: o.legal_name,
   403→    type: o.type,
   404→    country: o.country,
   405→    website: o.website,
   406→    verified: o.verified,
   407→    createdAt: o.created_at,
   408→    members: (membersByOrg.get(o.id) ?? []).map((m) => {
   409→      const u = memberUserBy.get(m.user_id)
   410→      return {
   411→        id: m.id,
   412→        userId: m.user_id,
   413→        name: u?.full_name || u?.email || m.user_id,
   414→        email: u?.email ?? '',
   415→        role: m.role,
   416→        title: m.title,
   417→        status: m.status,
   418→        lastLoginAt: u?.last_login_at ?? null,
   419→      }
   420→    }),
   421→  }))
   422→}

   443→export async function orderList(
   444→  actx: AdminContext,
   445→  opts: {status?: string; search?: string; priority?: string; assignedTo?: string; limit?: number} = {},
   446→) {
   447→  const assignedUserIds = actx.advisorId
   448→    ? unwrap<InvestorProfileRow[]>(
   449→        await supabase.from('investor_profiles').select('*').eq('advisor_id', actx.auth.user.id),
   450→        'profiles.list',
   451→      ).map((p) => p.user_id)
   452→    : null
   453→
   454→  let orderQuery = supabase.from('orders').select('*')
   455→  if (opts.status === 'open') orderQuery = orderQuery.in('status', OPEN_STATUSES)
   456→  else if (opts.status) orderQuery = orderQuery.eq('status', opts.status)
   457→  if (opts.priority) orderQuery = orderQuery.eq('priority', opts.priority)
   458→  if (opts.assignedTo === 'me') orderQuery = orderQuery.eq('assigned_to_id', actx.auth.user.id)
   459→  else if (opts.assignedTo === 'unassigned') orderQuery = orderQuery.is('assigned_to_id', null)
   460→  if (assignedUserIds) orderQuery = orderQuery.in('user_id', assignedUserIds)
   461→  const orderRows = unwrap<OrderRow[]>(
   462→    await orderQuery.order('created_at', {ascending: false}).limit(opts.limit ?? 400),
   463→    'orders.list',
   464→  )
   465→
   466→  const search = opts.search?.trim().toLowerCase()
   467→  let orders = orderRows.filter((o) => {
   468→    if (!search) return true
   469→    return (
   470→      o.reference.toLowerCase().includes(search) ||
   471→      (o.brief_title ?? '').toLowerCase().includes(search) ||
   472→      (o.contact_email ?? '').toLowerCase().includes(search) ||
   473→      (o.contact_name ?? '').toLowerCase().includes(search)
   474→    )
   475→  })
   476→
   477→  // relation include (payments) → separate query + JS join
   478→  const orderIds = orders.map((o) => o.id)
   479→  const paymentRows = orderIds.length
   480→    ? unwrap<PaymentRow[]>(
   481→        await supabase.from('payments').select('*').in('order_id', orderIds).order('created_at', {ascending: true}),
   482→        'payments.list',
   483→      )
   484→    : []
   485→  const paymentsBy = new Map<string, PaymentRow[]>()

   628→/**
   629→ * Advisor scope — OR(userId IN assigned, advisorId = actor) → two queries
   630→ * merged + deduped + re-sorted in JS (no multi-table tx over PostgREST).
   631→ */
   632→async function mandatesByAdvisorScope(
   633→  advisorUserId: string,
   634→  assignedUserIds: string[],
   635→  opts: {status?: string; stage?: string; ascending: boolean; limit: number},
   636→): Promise<MandateRow[]> {
   637→  const mkQuery = () => {
   638→    let q = supabase.from('mandates').select('*')
   639→    if (opts.status) q = q.eq('status', opts.status)
   640→    if (opts.stage) q = q.eq('stage', opts.stage)
   641→    return q
   642→  }
   643→  const [byUsers, byAdvisor] = await Promise.all([
   644→    unwrap<MandateRow[]>(
   645→      await mkQuery().in('user_id', assignedUserIds).order('updated_at', {ascending: opts.ascending}).limit(opts.limit),
   646→      'mandates.list',
   647→    ),
   648→    unwrap<MandateRow[]>(
   649→      await mkQuery().eq('advisor_id', advisorUserId).order('updated_at', {ascending: opts.ascending}).limit(opts.limit),
   650→      'mandates.list',
   651→    ),
   652→  ])
   653→  const merged: MandateRow[] = []
   654→  const seen = new Set<string>()
   655→  for (const m of [...byUsers, ...byAdvisor]) {
   656→    if (seen.has(m.id)) continue
   657→    seen.add(m.id)
   658→    merged.push(m)
   659→  }
   660→  merged.sort((a, b) => {
   661→    if (a.updated_at === b.updated_at) return 0
   662→    return (a.updated_at < b.updated_at) === opts.ascending ? -1 : 1
   663→  })
   664→  return merged.slice(0, opts.limit)
   665→}

   668→  actx: AdminContext,
   669→  opts: {status?: string; stage?: string; search?: string; limit?: number} = {},
   670→) {

   681→      status: opts.status,

   717→  const userBy = new Map(userRows.map((u) => [u.id, u]))
   718→  const advisorRows = unwrap<UserRow[]>(await supabase.from('users').select('*').in('role', ['advisor', 'admin']), 'users.list')
   719→  const advisorNames = new Map(advisorRows.map((a) => [a.id, a.full_name || a.email]))
   720→
   721→  return mandates.map((m) => ({
   722→    id: m.id,
   723→    reference: m.reference,
   724→    title: m.title,
   725→    status: m.status,
   726→    stage: m.stage,
   727→    probability: m.probability,

   730→    targetAllocationMin: m.targetAllocationMin,
   731→    targetAllocationMax: m.targetAllocationMax,
   732→    currency: m.currency,
   733→    geographies: m.geographies,
   734→    assetClasses: m.assetClasses,
   735→    minNetYield: m.minNetYield,
   736→    targetIrr: m.targetIrr,
   737→    investorName: userBy.get(m.userId)?.full_name || userBy.get(m.userId)?.email || m.userId,
   738→    investorEmail: userBy.get(m.userId)?.email ?? '',
   739→    advisorId: m.advisorId,
   740→    advisorName: m.advisorId ? advisorNames.get(m.advisorId) ?? null : null,
   741→    milestones: (milestonesBy.get(m.id) ?? []).map((ms) => ({id: ms.id, title: ms.title, status: ms.status, dueDate: ms.due_date})),

   744→    activatedAt: m.activatedAt,
   745→    closedAt: m.closedAt,

  1009→

  1012→    const cur = viewsBy.get(v.sanity_id) ?? {views: 0, viewers: new Set<string>()}

  1014→    cur.viewers.add(v.user_id)
  1015→    viewsBy.set(v.sanity_id, cur)
  1016→  }
  1017→  const savesBy = new Map<string, number>()
  1018→  for (const s of savedRows) savesBy.set(s.sanity_id, (savesBy.get(s.sanity_id) ?? 0) + 1)

  1020→  for (const o of orderRows) ordersBy.set(o.sanity_opportunity_id!, (ordersBy.get(o.sanity_opportunity_id!) ?? 0) + 1)

  1022→  for (const e of revenueEventRows) {
  1023→    const oppId = revenueOrderById.get(e.order_id)?.sanity_opportunity_id

  1025→    revenueBy.set(oppId, (revenueBy.get(oppId) ?? 0) + e.gross_usd)

  1039→      title: op?.title ?? meta?.title ?? id,

  1042→      assetClass: op?.assetClass ?? null,
  1043→      geography: op?.geography ?? null,
  1044→      indicativePrice: op?.indicativePrice ?? null,
  1045→      investmentSizeMin: op?.investmentSizeMin ?? null,
  1046→      investmentSizeMax: op?.investmentSizeMax ?? null,
  1047→      publishDate: op?.publishDate ?? null,
  1048→      auctionUrl: op?.auctionUrl ?? null,
  1049→      featuredInSanity: op?.featured ?? false,
  1050→      // governance meta
  1051→      metaStatus: meta?.status ?? null,
  1052→      metaTags: (meta?.tags as string[] | null) ?? [],
  1053→      metaVisibilityTier: meta?.visibilityTier ?? null,
  1054→      metaNotes: meta?.notes ?? null,
  1055→      // analytics
  1056→      views: analytics?.views ?? 0,
  1057→      uniqueViewers: analytics?.viewers.size ?? 0,
  1058→      saves: savesBy.get(id) ?? 0,
  1059→      orders: ordersBy.get(id) ?? 0,
  1060→      revenueUsd: Math.round((revenueBy.get(id) ?? 0) * 100) / 100,
  1061→    }
  1062→  })
  1063→
  1064→  return {rows, sanityConfigured: sanityOps.length > 0, studioUrl: sanityStudioUrl()}
