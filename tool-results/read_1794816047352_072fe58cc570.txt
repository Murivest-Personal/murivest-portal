    78→export async function investorList(
    79→  actx: AdminContext,
    80→  opts: {search?: string; tier?: string; verification?: string; classification?: string; dormant?: boolean; limit?: number} = {},
    81→): Promise<InvestorListRow[]> {
    82→  const scope = investorScopeWhere(actx)
    83→  const search = opts.search?.trim().toLowerCase() || null
    84→
    85→  const profiles = await db.investorProfile.findMany({
    86→    where: {
    87→      ...(scope ?? {}),
    88→      ...(opts.tier ? {accessTier: opts.tier} : {}),
    89→      ...(opts.verification ? {verificationStatus: opts.verification} : {}),
    90→      ...(opts.classification ? {classification: opts.classification} : {}),
    91→    },
    92→    include: {
    93→      user: {select: {id: true, email: true, fullName: true, status: true, lastLoginAt: true, createdAt: true}},
    94→    },
    95→    orderBy: {createdAt: 'desc'},
    96→    take: opts.limit ?? 500,
    97→  })
    98→
    99→  let rows = profiles.filter((p) => {
   100→    if (!search) return true
   101→    return (
   102→      p.user.email.toLowerCase().includes(search) ||
   103→      (p.user.fullName ?? '').toLowerCase().includes(search) ||
   104→      (p.countryResidence ?? '').toLowerCase().includes(search)
   105→    )
   106→  })
   107→
   108→  if (opts.dormant) {
   109→    const cutoff = Date.now() - 60 * 86400000
   110→    rows = rows.filter((p) => !p.user.lastLoginAt || p.user.lastLoginAt.getTime() < cutoff)
   111→  }
   112→
   113→  const userIds = rows.map((r) => r.userId)
   114→  const [orders, revenueEvents, mandates, orgs] = await Promise.all([
   115→    userIds.length ? db.order.findMany({where: {userId: {in: userIds}}, select: {userId: true}}) : [],
   116→    userIds.length ? db.revenueEvent.findMany({select: {grossUsd: true, order: {select: {userId: true}}}}) : [],
   117→    userIds.length ? db.mandate.findMany({where: {userId: {in: userIds}, status: 'active'}, select: {userId: true}}) : [],
   118→    db.organisation.findMany({select: {id: true, name: true}}),
   119→  ])
   120→  const orderCount = new Map<string, number>()
   121→  for (const o of orders) orderCount.set(o.userId, (orderCount.get(o.userId) ?? 0) + 1)
   122→  const revenueBy = new Map<string, number>()
   123→  for (const e of revenueEvents) revenueBy.set(e.order.userId, (revenueBy.get(e.order.userId) ?? 0) + e.grossUsd)
   124→  const mandateCount = new Map<string, number>()
   125→  for (const m of mandates) mandateCount.set(m.userId, (mandateCount.get(m.userId) ?? 0) + 1)
   126→  const orgNames = new Map(orgs.map((o) => [o.id, o.name]))
   127→
   128→  return rows.map((p) => ({
   129→    userId: p.userId,
   130→    email: p.user.email,
   131→    fullName: p.user.fullName,
   132→    status: p.user.status,
   133→    classification: p.classification,
   134→    verificationStatus: p.verificationStatus,
   135→    kycStatus: p.kycStatus,
   136→    amlStatus: p.amlStatus,
   137→    accessTier: p.accessTier,
   138→    ndaAccepted: p.ndaAccepted,
   139→    advisorId: p.advisorId,
   140→    source: p.source,
   141→    countryResidence: p.countryResidence,
   142→    organisationId: p.organisationId,
   143→    organisationName: p.organisationId ? orgNames.get(p.organisationId) ?? null : null,
   144→    equityAvailable: p.equityAvailable,
   145→    totalCapacity: p.totalCapacity,
   146→    capitalFloor: p.capitalFloor,
   147→    capitalCeiling: p.capitalCeiling,
   148→    capitalCurrency: p.capitalCurrency,
   149→    createdAt: p.user.createdAt,
   150→    lastLoginAt: p.user.lastLoginAt,
   151→    orderCount: orderCount.get(p.userId) ?? 0,
   152→    revenueUsd: Math.round((revenueBy.get(p.userId) ?? 0) * 100) / 100,
   153→    activeMandates: mandateCount.get(p.userId) ?? 0,
   154→  }))

   243→export async function orderList(
   244→  actx: AdminContext,
   245→  opts: {status?: string; search?: string; priority?: string; assignedTo?: string; limit?: number} = {},
   246→) {
   247→  const assignedUserIds = actx.advisorId
   248→    ? (await db.investorProfile.findMany({where: {advisorId: actx.auth.user.id}, select: {userId: true}})).map((p) => p.userId)
   249→    : null
   250→
   251→  const where: Record<string, unknown> = {}
   252→  if (opts.status === 'open') where.status = {in: OPEN_STATUSES}
   253→  else if (opts.status) where.status = opts.status
   254→  if (opts.priority) where.priority = opts.priority
   255→  if (opts.assignedTo === 'me') where.assignedToId = actx.auth.user.id
   256→  else if (opts.assignedTo === 'unassigned') where.assignedToId = null
   257→  if (assignedUserIds) where.userId = {in: assignedUserIds}
   258→
   259→  let orders = await db.order.findMany({
   260→    where,
   261→    orderBy: [{createdAt: 'desc'}],
   262→    take: opts.limit ?? 400,
   263→    include: {payments: {select: {id: true, status: true, method: true, amountUsd: true, amountKes: true, reference: true, createdAt: true, verifiedAt: true}}},
   264→  })
   265→
   266→  const search = opts.search?.trim().toLowerCase()
   267→  if (search) {
   268→    orders = orders.filter(
   269→      (o) =>
   270→        o.reference.toLowerCase().includes(search) ||
   271→        o.briefTitle.toLowerCase().includes(search) ||
   272→        (o.contactEmail ?? '').toLowerCase().includes(search) ||
   273→        (o.contactName ?? '').toLowerCase().includes(search),
   274→    )
   275→  }
   276→
   277→  const userIds = [...new Set(orders.map((o) => o.userId))]
   278→  const profiles = userIds.length
   279→    ? await db.investorProfile.findMany({where: {userId: {in: userIds}}, select: {userId: true, classification: true, accessTier: true, advisorId: true, organisationId: true}})
   280→    : []
   281→  const users = userIds.length ? await db.user.findMany({where: {id: {in: userIds}}, select: {id: true, fullName: true, email: true}}) : []
   282→  const orgIds = [...new Set(profiles.map((p) => p.organisationId).filter((x): x is string => !!x))]
   283→  const orgs = orgIds.length ? await db.organisation.findMany({where: {id: {in: orgIds}}, select: {id: true, name: true}}) : []
   284→  const profileBy = new Map(profiles.map((p) => [p.userId, p]))
   285→  const userBy = new Map(users.map((u) => [u.id, u]))
   286→  const orgBy = new Map(orgs.map((o) => [o.id, o.name]))
   287→
   288→  return orders.map((o) => ({
   289→    id: o.id,
   290→    reference: o.reference,
   291→    userId: o.userId,
   292→    investorName: userBy.get(o.userId)?.fullName || userBy.get(o.userId)?.email || o.userId,
   293→    investorEmail: userBy.get(o.userId)?.email ?? '',
   294→    investorClass: profileBy.get(o.userId)?.classification ?? 'individual',
   295→    investorTier: profileBy.get(o.userId)?.accessTier ?? 'tier_1',
   296→    organisationName: profileBy.get(o.userId)?.organisationId ? orgBy.get(profileBy.get(o.userId)!.organisationId!) ?? null : null,
   297→    briefTitle: o.briefTitle,
   298→    briefType: o.briefType,
   299→    briefSource: o.briefSource,
   300→    format: o.format,
   301→    customScope: o.customScope,
   302→    status: o.status,
   303→    priority: o.priority,
   304→    assignedToId: o.assignedToId,
   305→    advisorId: o.advisorId ?? profileBy.get(o.userId)?.advisorId ?? null,
   306→    amountUsd: o.amountUsd,
   307→    amountKes: o.amountKes,
   308→    currency: o.currency,
   309→    createdAt: o.createdAt,
   310→    dueAt: o.dueAt,
   311→    deliveredAt: o.deliveredAt,
   312→    closedAt: o.closedAt,
   313→    payment: o.payments[0] ?? null,
   314→    hasVerifiedPayment: o.payments.some((p) => p.status === 'verified'),
   315→  }))
   316→}

   318→export async function orderDetail(actx: AdminContext, orderId: string) {
   319→  const order = await db.order.findUnique({
   320→    where: {id: orderId},
   321→    include: {payments: true},
   322→  })
   323→  if (!order) return null
   324→
   325→  const [profile, user, events, notes, refunds, revenue] = await Promise.all([
   326→    db.investorProfile.findUnique({where: {userId: order.userId}}),
   327→    db.user.findUnique({where: {id: order.userId}, select: {id: true, fullName: true, email: true, status: true}}),
   328→    db.orderStatusEvent.findMany({where: {orderId}, orderBy: {createdAt: 'asc'}}),
   329→    db.internalNote.findMany({where: {entityType: 'order', entityId: orderId}, orderBy: {createdAt: 'desc'}}),
   330→    db.refundAdjustment.findMany({where: {orderId}, orderBy: {createdAt: 'desc'}}),
   331→    db.revenueEvent.findUnique({where: {orderId}}),
   332→  ])
   333→
   334→  if (actx.advisorId && profile?.advisorId !== actx.auth.user.id) return null
   335→
   336→  const staffIds = [
   337→    ...new Set(
   338→      [
   339→        ...events.map((e) => e.changedById),
   340→        ...notes.map((n) => n.authorId),
   341→        ...refunds.map((r) => r.requestedById),
   342→        ...refunds.map((r) => r.reviewedById ?? null),
   343→        order.assignedToId,
   344→        order.advisorId,
   345→        profile?.advisorId ?? null,
   346→      ].filter((x): x is string => !!x),
   347→    ),
   348→  ]

   351→
   352→  const assignments = await db.order.findUnique({where: {id: orderId}, select: {assignedToId: true, advisorId: true}})
   353→
   354→  return {
   355→    order,
   356→    investor: {user, profile},
   357→    payments: order.payments,
   358→    events: events.map((e) => ({...e, changedByName: e.changedById ? staffNames.get(e.changedById) ?? 'Staff' : 'System'})),

   361→      ...r,
   362→      requestedByName: staffNames.get(r.requestedById) ?? 'Staff',
   363→      reviewedByName: r.reviewedById ? staffNames.get(r.reviewedById) ?? 'Staff' : null,
   364→    })),
   365→    revenue,
   366→    assignment: {
   367→      assignedToName: assignments?.assignedToId ? staffNames.get(assignments.assignedToId) ?? null : null,
   368→      advisorName: (assignments?.advisorId ?? profile?.advisorId ?? null) ? staffNames.get(assignments?.advisorId ?? profile!.advisorId!) ?? null : null,
   369→    },
   370→  }
