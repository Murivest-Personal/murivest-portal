     2→     2→-- MURIVEST PRIVATE CAPITAL PORTAL — 001_schema.sql
     3→     3→-- Production PostgreSQL schema (PostgreSQL 14+; Supabase-compatible).
     4→     4→--
     5→     5→-- Creates every table backing each interaction with the portal:
     6→     6→--   • authentication (users, sessions, tokens, security events)
     7→     7→--   • organisations & membership
     8→     8→--   • investor profiles, verification pipeline & criteria
     9→     9→--   • interaction with Sanity-sourced content (saved / viewed opportunities,
    10→    10→--     saved searches)
    11→    11→--   • Investment Studio (scenarios, comparisons)
    12→    12→--   • Orders & payments (private / institutional briefs)
    13→    13→--   • Mandates & milestones
    14→    14→--   • Communications (enquiries, messages, notifications)
    15→    15→--   • Secure data room (document grants, access logs, requests, NDA)
    16→    16→--   • Portfolio & capital calls
    17→    17→--   • Audit trail
    18→    18→--
    19→    19→-- Content itself (opportunities, briefs, documents, market reports) is
    20→    20→-- authored in Sanity and fetched via GROQ; local tables reference content
    21→    21→-- by Sanity document id (`sanity_*` columns).
    22→    22→--
    23→    23→-- Ids are TEXT (cuid generated application-side). Apply with psql:
    24→    24→--   psql "$DATABASE_URL" -f sql/001_schema.sql
    25→    25→--   psql "$DATABASE_URL" -f sql/002_rls.sql
    26→    26→--   psql "$DATABASE_URL" -f sql/003_functions.sql
    27→    27→-- =============================================================================

   543→   543→-- Shared triggers: updated_at maintenance + critical-action audit capture
   544→   544→-- -----------------------------------------------------------------------------
   545→   545→CREATE OR REPLACE FUNCTION portal.touch_updated_at() RETURNS trigger AS $$
   546→   546→BEGIN
   547→   547→  NEW.updated_at := now();
   548→   548→  RETURN NEW;
   549→   549→END; $$ LANGUAGE plpgsql;
   550→   550→
   551→   551→-- Auto-audit for critical tables (orders, payments, document grants, profiles)
   552→   552→CREATE OR REPLACE FUNCTION audit.log_change() RETURNS trigger AS $$
   553→   553→DECLARE
   554→   554→  v_actor TEXT;
   555→   555→  v_entity TEXT;
   556→   556→  v_entity_id TEXT;
   557→   557→BEGIN
   558→   558→  v_entity   := TG_ARGV[0];
   559→   559→  v_actor    := NULLIF(current_setting('app.actor_id', true), '');
   560→   560→  v_entity_id := COALESCE(NEW.id, OLD.id);
   561→   561→  INSERT INTO audit.logs (id, actor_id, actor_type, action, entity, entity_id, detail)
   562→   562→  VALUES (
   563→   563→    'audit-' || replace(gen_random_uuid()::text, '-', ''),
   564→   564→    v_actor,
   565→   565→    CASE WHEN v_actor IS NULL THEN 'system'::portal.actor_type ELSE 'user'::portal.actor_type END,
   566→   566→    TG_ARGV[0] || '.' || TG_OP::text,
   567→   567→    v_entity,
   568→   568→    v_entity_id,
   569→   569→    jsonb_build_object('operation', TG_OP, 'new', to_jsonb(CASE WHEN TG_OP = 'DELETE' THEN NULL ELSE NEW END), 'old', to_jsonb(CASE WHEN TG_OP = 'UPDATE' THEN OLD END))
   570→   570→  );
   571→   571→  RETURN COALESCE(NEW, OLD);
   572→   572→END; $$ LANGUAGE plpgsql;
   573→   573→
   574→   574→DO $$
   575→   575→DECLARE t TEXT;
   576→   576→BEGIN
   577→   577→  FOREACH t IN ARRAY ARRAY[
   578→   578→    'users','organisations','investor_profiles','investor_criteria',
   579→   579→    'orders','payments','mandates','mandate_milestones',
   580→   580→    'document_grants','portfolio_positions','capital_calls',
   581→   581→    'scenarios','comparisons','enquiries','organisation_members'
   582→   582→  ]
   583→   583→  LOOP
   584→   584→    EXECUTE format('CREATE TRIGGER trg_%s_touch BEFORE UPDATE ON portal.%I
   585→   585→                   FOR EACH ROW EXECUTE FUNCTION portal.touch_updated_at()', t, t);
   586→   586→  END LOOP;
   587→   587→END; $$;
   588→   588→
   589→   589→CREATE TRIGGER trg_orders_audit    AFTER INSERT OR UPDATE OR DELETE ON portal.orders
   590→   590→  FOR EACH ROW EXECUTE FUNCTION audit.log_change('order');
   591→   591→CREATE TRIGGER trg_payments_audit AFTER INSERT OR UPDATE OR DELETE ON portal.payments
   592→   592→  FOR EACH ROW EXECUTE FUNCTION audit.log_change('payment');
   593→   593→CREATE TRIGGER trg_docgrants_audit AFTER INSERT OR DELETE ON portal.document_grants
   594→   594→  FOR EACH ROW EXECUTE FUNCTION audit.log_change('permission');
   595→   595→CREATE TRIGGER trg_profiles_audit AFTER UPDATE OF verification_status, kyc_status, access_tier, nda_accepted ON portal.investor_profiles
   596→   596→  FOR EACH ROW EXECUTE FUNCTION audit.log_change('profile');
   597→   597→
   598→   598→COMMIT;
